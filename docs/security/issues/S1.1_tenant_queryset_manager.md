# Issue S1.1: Tenant QuerySet & Manager Auto-Filtering

**Status**: 🔄 In Progress  
**Phase**: S1B (Infrastructure Implementation)  
**Scope**: Robust TenantManager with automatic organization filtering

## Functional Change

Replace `threading.local` (not async-safe) with `contextvars` (modern, concurrent, async-safe).

Implement auto-filtering: when `.objects` is used, TenantManager MUST inject `.filter(organization_id=current_org_id)` automatically.

**Before**:
```python
# TenantManager relies on thread-local (fragile, not async-safe)
class TenantManager(models.Manager):
    def get_queryset(self):
        org_id = get_current_organization()  # ← threading.local
        if org_id:
            return super().get_queryset().filter(organization_id=org_id)
        return super().get_queryset()  # ← LEAKS all records if context is None
```

**After**:
```python
# TenantManager with contextvars (safe, enforced)
class TenantManager(models.Manager):
    def get_queryset(self):
        org_id = get_current_organization()  # ← contextvars
        if org_id:
            return super().get_queryset().filter(organization_id=org_id)
        # FAIL-SAFE: No context → empty queryset (not all records)
        # Explicit context required; default-deny principle
        return super().get_queryset().none()
```

## Code Implementation

### Step 1: contextvars Backend (async-safe)

**File**: `src/infrastructure/multitenancy/context.py` (NEW)

```python
import contextvars
from typing import Optional
from uuid import UUID

_current_organization_context: contextvars.ContextVar[Optional[UUID]] = contextvars.ContextVar(
    'current_organization_id',
    default=None
)

def set_current_organization(org_id: Optional[UUID]) -> None:
    """Set organization context for current execution context (async-safe)."""
    _current_organization_context.set(org_id)

def get_current_organization() -> Optional[UUID]:
    """Get organization context (returns None if not set)."""
    return _current_organization_context.get()

def clear_current_organization() -> None:
    """Clear organization context."""
    _current_organization_context.set(None)

class TenantContextManager:
    """Context manager for explicit tenant scoping."""
    
    def __init__(self, org_id: UUID):
        self.org_id = org_id
        self.token = None
    
    def __enter__(self):
        self.token = _current_organization_context.set(self.org_id)
        return self
    
    def __exit__(self, *args):
        _current_organization_context.reset(self.token)
```

### Step 2: Robust TenantManager

**File**: `src/infrastructure/multitenancy/managers.py` (REPLACE)

```python
from django.db import models
from .context import get_current_organization

class TenantQuerySet(models.QuerySet):
    """QuerySet filtered by current organization context."""
    pass

class TenantManager(models.Manager):
    """Manager with automatic organization filtering (fail-safe)."""
    
    def get_queryset(self):
        qs = super().get_queryset()
        org_id = get_current_organization()
        
        if org_id:
            # Context set: filter automatically
            return qs.filter(organization_id=org_id)
        else:
            # Context NOT set: return empty queryset (explicit fail-safe)
            # Reason: Prevents accidental data leaks when context is missing
            return qs.none()
    
    def all_objects(self):
        """Bypass tenant filtering (admin/audit only)."""
        return super().get_queryset()
```

```python
from .multitenancy.managers import TenantManager, TenantQuerySet

class TenantModel(models.Model):
    organization = models.ForeignKey(
        'domain.Organization', 
        on_delete=models.CASCADE,
        related_name="%(class)s_items"
    )
    
    # Automatic filtering manager
    objects = TenantManager.from_queryset(TenantQuerySet)()
    
    # Unfiltered access (for admin/audit)
    unfiltered = models.Manager()
    
    class Meta:
        abstract = True
```

```python
    def __init__(self, get_response):
        self.get_response = get_response

        if org_id:
            try:
                org = Organization.objects.unfiltered.get(id=org_id)
            except (Organization.DoesNotExist, ValueError):
                pass

                except Organization.DoesNotExist:
                    pass

            if org:
                request.organization = org
```

### Unit Test: TenantManager Auto-Filtering

**File**: `src/tests/security/test_tenantmanager_autofilt.py`

```python
import pytest
from src.domain.models import Order
from src.infrastructure.multitenancy.context import (
    set_current_organization, clear_current_organization
)

@pytest.mark.django_db
class TestTenantManagerAutoFiltering:
    
    def test_context_set_filters_automatically(self, organization, order_a, order_b):
        """Order.objects filters by context automatically."""
        set_current_organization(organization.id)
        try:
            # Should see only order_a
            qs = Order.objects.all()
            assert qs.count() == 1
            assert qs.first().id == order_a.id
        finally:
            clear_current_organization()
    
    def test_context_none_returns_empty(self, order_a, order_b):
        """No context → empty queryset (fail-safe)."""
        clear_current_organization()
        
        qs = Order.objects.all()
        assert qs.count() == 0  # Empty, not leaked
    
    def test_unfiltered_returns_all(self, organization, order_a, order_b):
        """unfiltered bypasses context check."""
        qs = Order.unfiltered.all()
        assert qs.count() == 2  # All records
```
