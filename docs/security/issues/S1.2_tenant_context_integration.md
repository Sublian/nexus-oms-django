## Functional Change

### Part A: Celery Task Decorator (Close S1A Gap)

Wrap Celery tasks to enforce explicit organization_id parameter and set context before execution.

**Before**:
```python
@shared_task
def sync_single_invoice_task(self, entry_id):
    entry = InvoiceSyncQueue.all_objects.get(id=entry_id)
    org_id = entry.organization_id  # Extract manually
    # ← NO context set; UseCase works by accident via explicit filter
    result = InvoiceStatusQueryUseCase().execute(entry)
```

**After**:
```python
@tenant_task(organization_id_param=0)  # ← Explicit: param position 0
def sync_single_invoice_task(self, entry_id):
    # Context is already set by decorator
    # ← Safe: .objects queries respect context automatically
    entry = InvoiceSyncQueue.objects.get(id=entry_id)
    result = InvoiceStatusQueryUseCase().execute(entry)
```

## Code Implementation

### Step 1: Celery Task Decorator

**File**: `src/infrastructure/celery_utils.py` (NEW)

```python
from functools import wraps
from uuid import UUID
from celery import shared_task as celery_shared_task
from .multitenancy.context import set_current_organization, clear_current_organization

def tenant_task(*args, **kwargs):
    """
    Decorator for Celery tasks that require tenant context.
    
    Args:
        organization_id_param (int): Position of organization_id argument
        
    Usage:
        @tenant_task(organization_id_param=0)
        def my_task(self, org_id, ...):
            # org_id context is automatically set
    """
    def decorator(func):
        org_id_param = kwargs.get('organization_id_param', 0)
        
        # Apply celery.shared_task
        func = celery_shared_task(func)
        
        # Wrap with context manager
        @wraps(func)
        def wrapper(self, *task_args, **task_kwargs):
            org_id = task_args[org_id_param] if org_id_param < len(task_args) else None
            
            if not org_id:
                raise ValueError(
                    f"tenant_task requires organization_id at position {org_id_param}"
                )
            
            try:
                set_current_organization(org_id)
                return func.original(self, *task_args, **task_kwargs)
            finally:
                clear_current_organization()
        
        wrapper.original = func
        return wrapper
    
    return decorator
```

### Step 3: Test Base Class for Tenant Isolation

**File**: `src/tests/security/base.py` (NEW)

```python
import pytest
from django.test import TestCase
from src.infrastructure.multitenancy.context import (
    set_current_organization, clear_current_organization
)

class TenantIsolationTestCase(TestCase):
    """Base class for tests validating tenant isolation."""
    
    @classmethod
    def setUpClass(cls):
        super().setUpClass()
        # Create test tenants and data
        from src.domain.models import Organization
        
        cls.tenant_a = Organization.objects.create(
            name="Test Tenant A",
            slug="test-tenant-a"
        )
        cls.tenant_b = Organization.objects.create(
            name="Test Tenant B",
            slug="test-tenant-b"
        )
    
    def set_context(self, organization):
        """Set context for current test."""
        set_current_organization(organization.id)
    
    def clear_context(self):
        """Clear context."""
        clear_current_organization()
    
    def tearDown(self):
        """Always clean up context after each test."""
        self.clear_context()
        super().tearDown()
    
    def assert_tenant_isolation(self, tenant, queryset, expected_count):
        """Assert that queryset respects tenant context."""
        self.set_context(tenant)
        self.assertEqual(queryset.count(), expected_count,
                        f"Expected {expected_count} results for {tenant.name}")
```

### Step 4: Cross-Tenant Validation Tests

**File**: `src/tests/security/test_tenant_isolation.py` (NEW)

```python
import pytest
from src.tests.security.base import TenantIsolationTestCase
from src.domain.models import Order, Product

@pytest.mark.django_db
class TestCrossTenantBoundary(TenantIsolationTestCase):
    """Validate tenant data isolation (cross-tenant access fails)."""
    
    @classmethod
    def setUpClass(cls):
        super().setUpClass()
        
        # Create products for each tenant
        cls.product_a = Product.objects.create(
            organization=cls.tenant_a,
            name="Product A",
            sku="SKU-A",
            price=100.00
        )
        cls.product_b = Product.objects.create(
            organization=cls.tenant_b,
            name="Product B",
            sku="SKU-B",
            price=200.00
        )
        
        # Create orders for each tenant
        cls.order_a = Order.objects.create(
            organization=cls.tenant_a,
            customer_name="Customer A",
            customer_email="a@tenant-a.com"
        )
        cls.order_b = Order.objects.create(
            organization=cls.tenant_b,
            customer_name="Customer B",
            customer_email="b@tenant-b.com"
        )
    
    def test_tenant_a_cannot_see_tenant_b_orders(self):
        """Tenant A context: Order.objects should not return Tenant B orders."""
        self.set_context(self.tenant_a)
        
        # Query all orders in context of Tenant A
        orders = Order.objects.all()
        
        # Should only see Tenant A's order
        self.assertEqual(orders.count(), 1)
        self.assertEqual(orders.first().id, self.order_a.id)
        
        # Verify Tenant B's order is not accessible
        self.assertFalse(
            Order.objects.filter(id=self.order_b.id).exists(),
            "Tenant A should not see Tenant B's order"
        )
    
    def test_tenant_b_cannot_see_tenant_a_orders(self):
        """Tenant B context: Order.objects should not return Tenant A orders."""
        self.set_context(self.tenant_b)
        
        orders = Order.objects.all()
        
        self.assertEqual(orders.count(), 1)
        self.assertEqual(orders.first().id, self.order_b.id)
        
        self.assertFalse(
            Order.objects.filter(id=self.order_a.id).exists(),
            "Tenant B should not see Tenant A's order"
        )
    
    def test_no_context_returns_empty(self):
        """No context set: .objects returns empty (fail-safe)."""
        self.clear_context()
        
        # Should return empty, not all orders
        orders = Order.objects.all()
        self.assertEqual(orders.count(), 0)
    
    def test_unfiltered_manager_returns_all(self):
        """unfiltered manager bypasses context."""
        self.clear_context()
        
        # unfiltered should return both orders
        orders = Order.unfiltered.all()
        self.assertEqual(orders.count(), 2)
```
